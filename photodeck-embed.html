// --- OBSŁUGA INTERFEJSU DRAWERA Z LICZNIKIEM 24H --- document.addEventListener('DOMContentLoaded', function () { var tab = document.getElementById('pi-drawer-tab'); var panel = document.getElementById('pi-drawer-panel'); var closeBtn = document.getElementById('pi-drawer-close'); var closeBtnAlt = document.getElementById('pi-drawer-close-btn'); var backdrop = document.getElementById('pi-drawer-backdrop'); var dot = document.getElementById('pi-drawer-dot'); // Sprawdzanie stanu 24-godzinnego if (window.CURRENT_BATCH_ID && dot) { var storageKey = 'pi_archive_seen_time_' + window.CURRENT_BATCH_ID; var firstSeenTime = localStorage.getItem(storageKey); var now = new Date().getTime(); var twentyFourHours = 24 * 60 * 60 * 1000; // 24 godziny w milisekundach if (!firstSeenTime) { // Użytkownik widzi tę paczkę po raz pierwszy w życiu -> Zapisujemy czas i pokazujemy zielony punkt localStorage.setItem(storageKey, now); dot.classList.remove('hidden'); // Otwórz szufladę automatycznie na start if (window.piOpenDrawer) { setTimeout(window.piOpenDrawer, 600); } } else { // Użytkownik już tu był – sprawdzamy, czy minęły 24 godziny var elapsed = now - parseInt(firstSeenTime, 10); if (elapsed < twentyFourHours) { // Minęło mniej niż 24h -> Zielony punkt nadal świeci dot.classList.remove('hidden'); } else { // Minęło więcej niż 24h -> Punkt staje się biały (ukrywamy zielony) dot.classList.add('hidden'); } } } function openDrawer() { panel.classList.add('open'); backdrop.classList.add('active'); panel.setAttribute('aria-hidden', 'false'); tab.setAttribute('aria-expanded', 'true'); document.addEventListener('keydown', onKeydown); } function closeDrawer() { panel.classList.remove('open'); backdrop.classList.remove('active'); panel.setAttribute('aria-hidden', 'true'); tab.setAttribute('aria-expanded', 'false'); document.removeEventListener('keydown', onKeydown); } function onKeydown(ev) { if (ev.key === 'Escape') closeDrawer(); } tab.addEventListener('click', function () { if (panel.classList.contains('open')) closeDrawer(); else openDrawer(); }); closeBtn.addEventListener('click', closeDrawer); if (closeBtnAlt) closeBtnAlt.addEventListener('click', closeDrawer); backdrop.addEventListener('click', closeDrawer); window.piOpenDrawer = openDrawer; window.piCloseDrawer = closeDrawer; });
